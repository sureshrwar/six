# SIX guest root filesystem manifest
# ---------------------------------------------------------------------------
# This file is the single declarative description of what goes into the guest
# root filesystem image (disk/x86/root).  port/image/mkimage.sh reads it,
# builds a staging tree under fakeroot, and hands that tree to
# "mke2fs -d" to produce a revision-0 ext2 image that the Linux 2.0.11
# filesystem driver inside SIX can read.
#
# The layout matches the 2005 SPARC root disk (preserved in git tag
# "v2005-cvs"), bringing the x86 port to full parity with SPARC:
#
#   kernel init() -> /etc/init -> sh /etc/rc -> getty -> login -> /bin/sh
#
# Columns are whitespace separated.  Blank lines and #-comments are ignored.
#
#   dir   <path-on-image>  <octal-mode>
#   file  <path-on-image>  <octal-mode>  <source-path-relative-to-source-root>
#   node  <path-on-image>  <octal-mode>  c|b  <major>  <minor>
#   sym   <path-on-image>  -             <link-target>
# ---------------------------------------------------------------------------

# --- /dev (block major 3 = HD_MAJOR) ---------------------------------------
# The raw root disk.  The kernel has always registered this device --
# hd_init() does register_blkdev(3, "hd", ...) and ROOT_DEV is hardcoded to
# 3:0 by parse_root_dev() -- but mount_root() reaches it by fabricating an
# inode rather than by resolving a path, so nothing ever required a node
# here.  This one exists purely so that userland (/bin/dd) can read the
# disk directly.
#
# Mode 0600 is conventional rather than protective: everything in SIX runs
# as root.  Writing to this device while / is mounted read-write will
# corrupt the filesystem, exactly as it would on real hardware.
node  /dev/hda              0600  b  3  0

# The auxiliary disk.  Minor 64 rather than 1: the hd driver uses the top
# two bits of the minor as the drive number and the bottom six as the
# partition, so DEVICE_NR() in <linux/blk.h> is MINOR(dev) >> 6.
#
# The node is here unconditionally, but the disk behind it usually is not.
# disk/x86/aux_storage-1 is not built by default and is not checked in --
# run "make aux-image" to create one.  Without it check_root() leaves the
# drive unopened, hd_geninit() keeps NR_HD at 1, and hd_open() answers
# ENODEV.  A device node is a name, not an assertion that the hardware
# exists; /dev/hdb on a real system behaves the same way when the second
# drive is unplugged.
#
# Nothing mounts this.  The kernel deliberately has no opinion on where it
# belongs -- see check_root() in init/main.c -- so attaching it is a job
# for /bin/mount, from /etc/rc or from a shell.
node  /dev/hdb              0600  b  3  64
